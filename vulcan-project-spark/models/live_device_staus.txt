SELECT
    _id,
    org_id,
    device_id,
    org_device_id,
    subscription_id,
    device_name,
    manufacturer,
    model_type,
    serial_number,
    family,
    enclosure_type,
    category,
    platform,
    os_name,
    device_groups,
    licenses,
    device_labels,
    time_zone,
    status,
    warranty_status,
    cast(warranty_expiration_date as timestamp) as warranty_expiration_date,
    country_name,
    version,
    network,
    device_licensed,
    connection_id,
    cast(connection_timestamp as timestamp) as connection_timestamp,
    cast(last_seen_date as timestamp) as last_seen_date,
    cast(date_created as timestamp) as date_created,
    to_timestamp(from_unixtime(CAST(_nilus_load_id AS DOUBLE))) AS _nilus_load_id,
    _nilus_id                                         
FROM (
    SELECT
        *,
        ROW_NUMBER() OVER (
            PARTITION BY _id
            ORDER BY CAST(_nilus_load_id AS BIGINT) DESC
        ) AS rn
    FROM s3depot.demolake04.live_device_status
)
WHERE rn = 1
